# Оптимізована nginx конфігурація для максимальної продуктивності
# Додайте ці налаштування в секцію server

server {
    listen 443 ssl http2;
    server_name landscaper.co.ua www.landscaper.co.ua;

    # SSL налаштування (ваші існуючі)
    ssl_certificate /etc/letsencrypt/live/landscaper.co.ua/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/landscaper.co.ua/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;

    # ⚡ ПРОДУКТИВНІСТЬ - Збільшуємо ліміти
    client_max_body_size 10M;
    client_body_buffer_size 128k;
    client_body_timeout 60s;
    client_header_timeout 60s;
    
    # ⚡ ПРОДУКТИВНІСТЬ - Оптимізація буферів
    proxy_buffering on;
    proxy_buffer_size 128k;
    proxy_buffers 4 256k;
    proxy_busy_buffers_size 256k;
    
    # 🗜️ СТИСНЕННЯ - Gzip компресія
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types
        text/plain
        text/css
        text/xml
        text/javascript
        application/json
        application/javascript
        application/xml+rss
        application/atom+xml
        image/svg+xml;

    # 📦 КЕШУВАННЯ - Статичні файли
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|pdf|txt|webp|avif)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header Vary Accept-Encoding;
        
        # Оптимізація для зображень
        location ~* \.(jpg|jpeg|png|gif|webp|avif)$ {
            expires 1M;
            add_header Cache-Control "public, immutable";
        }
    }
    
    # 🚀 КЕШУВАННЯ - Next.js статичні файли
    location /_next/static/ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # 📄 КЕШУВАННЯ - HTML файли
    location ~* \.(html|htm)$ {
        expires 5m;
        add_header Cache-Control "public, must-revalidate";
    }
    
    # 🔧 API КЕШУВАННЯ - Короткий кеш для API
    location /api/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Збільшені таймаути для API
        proxy_connect_timeout 120s;
        proxy_send_timeout 120s;
        proxy_read_timeout 120s;
        
        # Кеш для GET запитів API
        proxy_cache_methods GET HEAD;
        proxy_cache_valid 200 5m;
        proxy_cache_valid 404 1m;
        add_header X-Cache-Status $upstream_cache_status;
        
        # Дозволяємо великі запити для API
        client_max_body_size 10M;
    }

    # 🏠 ГОЛОВНА СТОРІНКА - Проксування на Next.js
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        
        # Оптимізовані таймаути
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
        
        # Кешування для статичних сторінок
        proxy_cache_methods GET HEAD;
        proxy_cache_valid 200 5m;
        proxy_cache_valid 404 1m;
        
        # Не кешуємо динамічні сторінки
        proxy_cache_bypass $cookie_session $http_authorization;
        proxy_no_cache $cookie_session $http_authorization;
    }

    # 🔒 БЕЗПЕКА - Заголовки (ваші існуючі + нові)
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Frame-Options DENY always;
    add_header X-Content-Type-Options nosniff always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    
    # 🚫 БЛОКУВАННЯ - Заборонені файли
    location ~ /\. {
        deny all;
    }
    
    location ~* \.(env|log|htaccess)$ {
        deny all;
    }
}

# 🗂️ ДОДАТКОВІ НАЛАШТУВАННЯ для nginx.conf (в секції http)

# Оптимізація worker процесів
worker_processes auto;
worker_connections 1024;

# Оптимізація файлових операцій
sendfile on;
tcp_nopush on;
tcp_nodelay on;

# Таймаути
keepalive_timeout 65;
keepalive_requests 100;

# Кеш для проксі
proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=my_cache:10m max_size=1g 
                 inactive=60m use_temp_path=off;

# Ліміти швидкості (опціонально)
limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
limit_req_zone $binary_remote_addr zone=login:10m rate=5r/s;
